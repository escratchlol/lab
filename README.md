# Домашнее задание к работе 9

## Условие задачи
Написать программу, которая выводит на экран геометрическую фигуру (четырёхугольник), отрисованную с помощью заданного пользователем символа. 
Фигура должна быть полой внутри (выводится только контур).
Размеры фигуры задаются пользователем: длины диагоналей d1 и d2 определяют размеры четырёхугольника в символах (ширина и высота), а угол α задает угол пересечения диагоналей. 
Предусмотреть возможность повторного запуска программы.
## 1. Алгоритм и блок-схема

### **Алгоритм**
1. **Начало**
2. Подключить необходимые библиотеки:
    - `<stdio.h>` - для ввода и вывода данных (`printf`, `scanf`).
   - `<locale.h>` - для работы с локалью (поддержка русского языка).
   - `<math.h>` — для математических функций (`sin`, `cos`, `fmax`, `fmin`, `ceil`, `floor`).
3. Определить константы `PI` и объявить прототипы вспомогательных функций:
   - `double vx(...)`, `double vy(...)` — вычисление координат вершин четырёхугольника;
   - `int in_quad(...)` — проверка, принадлежит ли точка внутренней области фигуры;
   - `void draw_figure(...)` — отрисовка фигуры в консоли.
4. Определить функцию `main()` - точку входа в программу:
   - Установить локаль для корректного отображения кириллицы: `setlocale(LC_ALL, "Ru")`.
   - Организовать бесконечный цикл while (1) для возможности повторного запуска:
       - Запросить у пользователя символ для отрисовки (`sym`), длины диагоналей (`d1`, `d2`) и угол между ними в градусах (`ang`).
       - Вызвать функцию `draw_figure(sym, d1, d2, ang)`.
       - Запросить у пользователя, продолжать ли работу. Если введен `0`, выйти из цикла с помощью `break`.
   - Завершить программу, вернув 0.
5. В теле функции `draw_figure`:
   - Проверить корректность введенных данных (диагонали > 0, угол от 0 до 180). Если данные некорректны — вывести сообщение об ошибке и завершить функцию (`return`).
   - Вычислить минимальные и максимальные координаты X и Y для определения границ отрисовки.
   - Запустить вложенные циклы для перебора строк (от `ymax` до `ymin`) и столбцов (от `xmin * 2` до `xmax * 2`). Шаг по X равен 0.5 для компенсации вытянутости символов по вертикали.
   - Если текущая точка (с учетом масштабирования) принадлежит фигуре (функция `in_quad`), вывести заданный символ, иначе — пробел.
6. В теле функции `in_quad`:
   - Реализовать алгоритм проверки принадлежности точки выпуклому многоугольнику через векторные произведения. Точка считается внутри, если она находится по одну сторону от всех четырёх рёбер.
7. **Конец**

## 2. Реализация программы

#include <stdio.h>
#include <locale.h>
#include <math.h>

#define PI 3.14159265358979

double vx(int k, double d1, double d2, double a);
double vy(int k, double d1, double d2, double a);
int in_quad(double x, double y, double d1, double d2, double a);
void draw_figure(char sym, double d1, double d2, double ang);

int main()
{
	setlocale(LC_ALL, "Ru");

	puts("Задание 3 (четырёхугольник по диагоналям и углу):");
	while (1)
	{
		char sym;
		double d1, d2, ang;
		printf("Введите символ, из которого будет строиться фигура: ");
		scanf(" %c", &sym);
		printf("Введите d1: ");
		scanf("%lf", &d1);
		printf("Введите d2: ");
		scanf("%lf", &d2);
		printf("Введите угол между диагоналями (в градусах): ");
		scanf("%lf", &ang);

		draw_figure(sym, d1, d2, ang);

		int cont;
		printf("\nПродолжить? (Да - введите любое число, Нет - введите 0): ");
		scanf("%d", &cont);
		if (cont == 0)
		{
			break;
		}
	}

	return 0;
}

void draw_figure(char sym, double d1, double d2, double ang)
{
	if (d1 <= 0 || d2 <= 0 || ang <= 0 || ang >= 180)
	{
		puts("ошибка: d1 и d2 должны быть больше 0, угол от 0 до 180");
		return;
	}

	double ymax = fmax(vy(2, d1, d2, ang), 0);
	double ymin = fmin(vy(4, d1, d2, ang), 0);
	double xmax = fmax(vx(3, d1, d2, ang), vx(2, d1, d2, ang));
	double xmin = fmin(vx(1, d1, d2, ang), vx(4, d1, d2, ang));

	printf("\n");
	for (int r = (int)ceil(ymax); r >= (int)floor(ymin); r--)
	{
		for (int cx = (int)floor(xmin * 2); cx <= (int)ceil(xmax * 2); cx++)
		{
			if (in_quad(cx * 0.5, r, d1, d2, ang))
			{
				printf("%c", sym);
			}
			else
			{
				printf(" ");
			}
		}
		printf("\n");
	}
}

double vx(int k, double d1, double d2, double a)
{
	double rad = a * PI / 180;
	switch (k)
	{
	case 1:
		return -0.4 * d1;
	case 2:
		return 0.6 * d2 * cos(rad);
	case 3:
		return 0.6 * d1;
	default:
		return -0.4 * d2 * cos(rad);
	}
}

double vy(int k, double d1, double d2, double a)
{
	double rad = a * PI / 180;
	switch (k)
	{
	case 2:
		return 0.6 * d2 * sin(rad);
	case 4:
		return -0.4 * d2 * sin(rad);
	default:
		return 0;
	}
}

int in_quad(double x, double y, double d1, double d2, double a)
{
	int pos = 0, neg = 0;
	for (int k = 1; k <= 4; k++)
	{
		int next = (k == 4) ? 1 : k + 1;
		double x1 = vx(k, d1, d2, a), y1 = vy(k, d1, d2, a);
		double x2 = vx(next, d1, d2, a), y2 = vy(next, d1, d2, a);
		double cross = (x2 - x1) * (y - y1) - (y2 - y1) * (x - x1);
		if (cross > 1e-9)
		{
			pos = 1;
		}
		if (cross < -1e-9)
		{
			neg = 1;
		}
	}
	return !(pos && neg);
}
