# Домашнее задание к работе 9

## Условие задачи
Написать программу, которая выводит на экран геометрическую фигуру (четырёхугольник), отрисованную с помощью заданного пользователем символа. 
Фигура должна быть полой внутри (выводится только контур).
Размеры фигуры задаются пользователем: длины диагоналей d1 и d2 определяют размеры четырёхугольника в символах (ширина и высота), а угол α задает угол пересечения диагоналей. 
Предусмотреть возможность повторного запуска программы.
## 1. Алгоритм и блок-схема

### **Алгоритм**
1. **Начало**
2. становить локаль для корректного отображения кириллицы: setlocale(LC_ALL, "Ru").
3. Вывести заголовок: «задание 3 (четырёхугольник по диагоналям и углу):».
4. Организовать бесконечный цикл while(1):
   - Ввести исходные данные: символ sym, длины диагоналей d1, d2 и угол ang (в градусах).
   - Если param_ok(d1, d2, ang) ложно (возвращает 0):
     	- Вывести сообщение об ошибке.
   - Иначе (если параметры корректны):
     	- Вычислить границы отрисовки: ymax, ymin, xmax, xmin (используя вызовы vx и vy для вершин 1..4).
     	- Вывести \n (перевод строки).
     	- Цикл по строкам r от ceil(ymax) до floor(ymin) (шаг −1):
     	  	- Цикл по столбцам cx от floor(xmin * 2) до ceil(xmax * 2) (шаг +1):
     	  	  	- Вызвать f_char(cx * 0.5, r, d1, d2, ang, sym) и вывести возвращённый символ.
     	  	- Конец цикла по cx, вывести \n.
   - Запросить cont (продолжить?).
   - Если cont == 0, выйти из цикла while(1).
5. **Конец**

**Часть 2: Функция проверки параметров (param_ok)**
1. **Начало функции**: получение d1, d2, ang.
2. Если d1≤0 или d2≤0 или ang <= 0 или ang ≥ 180, вернуть 0 (ошибка).
3. Иначе вернуть 1 (успех).
4. **Конец**.
   
**Часть 3: функция vx (x-координата вершины)**
1. **Начало функции**: получение k (номер вершины 1..4), d1, d2, a.
2. Перевести угол в радианы: rad = a * PI / 180.
3. В зависимости от k:
   - k = 1: вернуть -0.4 * d1;
   - k = 2: вернуть 0.6 * d2 * cos(rad);
   - k = 3: вернуть 0.6 * d1;
   - k = 4 (по умолчанию): вернуть -0.4 * d2 * cos(rad).
4. **Конец**.

**Часть 4: функция vy (y-координата вершины)**
1. **Начало функции**: получение k, d1, d2, a.
2. Перевести угол в радианы: rad = a * PI / 180.
3. В зависимости от k:
   - k = 2: вернуть 0.6 * d2 * sin(rad);
   - k = 4: вернуть -0.4 * d2 * sin(rad);
   - иначе (k = 1 или 3): вернуть 0.
4. **Конец**.

**Часть 5: функция f_char (выбор символа для точки)**
1. **Начало функции**: получение x, y, d1, d2, a, sym.
2. Для каждого ребра k от 1 до 4:
   - определить следующую вершину next;
   - получить координаты вершин (x1, y1) и (x2, y2);
   - вычислить расстояние от точки до ребра с учётом масштабирования по X (умножение на 2 для компенсации вытянутости символов):
     - ddx = (x - x1) * 2.0, ddy = (y - y1);
     - sx = (x2 - x1) * 2.0, sy = (y2 - y1);
     - t = (ddx*sx + ddy*sy) / (sx*sx + sy*sy), ограничить t на [0, 1];
     - ex = ddx - t*sx, ey = ddy - t*sy;
     - если sqrt(ex*ex + ey*ey) < 0.5, вернуть sym.
3. Если ни для одного ребра условие не выполнено, вернуть пробел ' '.
4. **Конец**.

## 2. Реализация программы

```c
#include <stdio.h>
#include <locale.h>
#include <math.h>

#define PI 3.14159265358979

double vx(int k, double d1, double d2, double a);
double vy(int k, double d1, double d2, double a);
int in_quad(double x, double y, double d1, double d2, double a);
void draw_figure(char sym, double d1, double d2, double ang);

int main()
{
	setlocale(LC_ALL, "Ru");

	puts("задание 3 (четырёхугольник по диагоналям и углу):");
	while (1)
	{
		char sym;
		double d1, d2, ang;
		printf("Введите символ, из которого будет строиться фигура: ");
		scanf(" %c", &sym);
		printf("Введите d1: ");
		scanf("%lf", &d1);
		printf("Введите d2: ");
		scanf("%lf", &d2);
		printf("Введите угол между диагоналями (в градусах): ");
		scanf("%lf", &ang);

		draw_figure(sym, d1, d2, ang);

		int cont;
		printf("\nПродолжить? (Да - введите любое число, Нет - введите 0): ");
		scanf("%d", &cont);
		if (cont == 0)
		{
			break;
		}
	}

	return 0;
}

void draw_figure(char sym, double d1, double d2, double ang)
{
	if (d1 <= 0 || d2 <= 0 || ang <= 0 || ang >= 180)
	{
		puts("ошибка: d1 и d2 должны быть больше 0, угол от 0 до 180");
		return;
	}

	double ymax = fmax(vy(2, d1, d2, ang), 0);
	double ymin = fmin(vy(4, d1, d2, ang), 0);
	double xmax = fmax(vx(3, d1, d2, ang), vx(2, d1, d2, ang));
	double xmin = fmin(vx(1, d1, d2, ang), vx(4, d1, d2, ang));

	printf("\n");
	for (int r = (int)ceil(ymax); r >= (int)floor(ymin); r--)
	{
		for (int cx = (int)floor(xmin * 2); cx <= (int)ceil(xmax * 2); cx++)
		{
			if (in_quad(cx * 0.5, r, d1, d2, ang))
			{
				printf("%c", sym);
			}
			else
			{
				printf(" ");
			}
		}
		printf("\n");
	}
}

double vx(int k, double d1, double d2, double a)
{
	double rad = a * PI / 180;
	switch (k)
	{
	case 1:
		return -0.4 * d1;
	case 2:
		return 0.6 * d2 * cos(rad);
	case 3:
		return 0.6 * d1;
	default:
		return -0.4 * d2 * cos(rad);
	}
}

double vy(int k, double d1, double d2, double a)
{
	double rad = a * PI / 180;
	switch (k)
	{
	case 2:
		return 0.6 * d2 * sin(rad);
	case 4:
		return -0.4 * d2 * sin(rad);
	default:
		return 0;
	}
}

int in_quad(double x, double y, double d1, double d2, double a)
{
	int pos = 0, neg = 0;
	for (int k = 1; k <= 4; k++)
	{
		int next = (k == 4) ? 1 : k + 1;
		double x1 = vx(k, d1, d2, a), y1 = vy(k, d1, d2, a);
		double x2 = vx(next, d1, d2, a), y2 = vy(next, d1, d2, a);
		double cross = (x2 - x1) * (y - y1) - (y2 - y1) * (x - x1);
		if (cross > 1e-9)
		{
			pos = 1;
		}
		if (cross < -1e-9)
		{
			neg = 1;
		}
	}
	return !(pos && neg);
}
```
