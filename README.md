## 2. Реализация программы

#include <stdio.h>
#include <locale.h>
#include <math.h>

double sumN(int n);

int main()
{
    setlocale(LC_ALL, "Russian");

    puts("задание 3");
    int n1;
    printf("Введите n: ");
    scanf("%d", &n1);
    double sum_result = sumN(n1);
    printf("Сумма = %.6lf\n", sum_result);

    return 0;
}

double sumN(int n) {
    double sum = 0.0;
    double s_sin = 0.0;
    for (int i = 1; i <= n; i++) {
        s_sin += sin(i);
        sum += 1.0 / s_sin;
    }
    return sum;
}
